from __future__ import annotations

import os
from io import BytesIO

import pandas as pd
import streamlit as st

from tsukuba_bookmap.extract_references import aggregate_books, extract_references
from tsukuba_bookmap.ingest import read_uploaded_table, sample_courses, standardize_courses
from tsukuba_bookmap.lookup import DEFAULT_TSUKUBA_KID, lookup_references, merge_holdings
from tsukuba_bookmap.utils import csv_bytes

st.set_page_config(page_title="Tsukuba BookMap", page_icon="📚", layout="wide")


@st.cache_data(show_spinner=False)
def cached_standardize(raw: pd.DataFrame) -> pd.DataFrame:
    return standardize_courses(raw)


@st.cache_data(show_spinner=False)
def cached_extract(courses: pd.DataFrame) -> pd.DataFrame:
    return extract_references(courses)


def excel_bytes(sheets: dict[str, pd.DataFrame]) -> bytes:
    bio = BytesIO()
    with pd.ExcelWriter(bio, engine="openpyxl") as writer:
        for name, df in sheets.items():
            safe = name[:31]
            df.to_excel(writer, sheet_name=safe, index=False)
    return bio.getvalue()


def load_courses_from_sidebar() -> pd.DataFrame:
    st.sidebar.header("データ")
    uploaded = st.sidebar.file_uploader("KdB CSV/Excelをアップロード", type=["csv", "xlsx", "xls"])
    if uploaded is None:
        st.sidebar.info("未アップロード時はサンプルデータを表示します。")
        return sample_courses()
    raw = read_uploaded_table(uploaded, uploaded.name)
    return cached_standardize(raw)


def sidebar_settings() -> tuple[str, str, str]:
    st.sidebar.header("図書館照合")
    appid = st.sidebar.text_input("CiNiiアプリケーションID", value=os.environ.get("CINII_APPID", ""), type="password")
    kid = st.sidebar.text_input("筑波大学 機関ID", value=DEFAULT_TSUKUBA_KID, help="筑波大学全体で照合する場合は既定値のまま")
    fano = st.sidebar.text_input("個別図書館ID（任意）", value="", help="例: FA...。空欄なら機関IDで照合")
    return appid, kid, fano


def filter_courses_ui(courses: pd.DataFrame, prefix: str = "") -> pd.DataFrame:
    c1, c2, c3, c4 = st.columns(4)
    with c1:
        years = sorted([str(x) for x in courses["year"].dropna().unique() if str(x).strip()])
        year_sel = st.multiselect("年度", years, default=years, key=f"{prefix}year")
    with c2:
        orgs = sorted([str(x) for x in courses["organization"].dropna().unique() if str(x).strip()])
        org_sel = st.multiselect("開設組織", orgs, default=[], key=f"{prefix}org")
    with c3:
        terms = sorted([str(x) for x in courses["term"].dropna().unique() if str(x).strip()])
        term_sel = st.multiselect("開講時期", terms, default=[], key=f"{prefix}term")
    with c4:
        keyword = st.text_input("科目名・教員・科目番号", key=f"{prefix}keyword")

    filtered = courses.copy()
    if year_sel:
        filtered = filtered[filtered["year"].astype(str).isin(year_sel)]
    if org_sel:
        filtered = filtered[filtered["organization"].astype(str).isin(org_sel)]
    if term_sel:
        filtered = filtered[filtered["term"].astype(str).isin(term_sel)]
    if keyword:
        k = keyword.lower()
        mask = (
            filtered["course_name"].astype(str).str.lower().str.contains(k, regex=False)
            | filtered["instructor"].astype(str).str.lower().str.contains(k, regex=False)
            | filtered["course_number"].astype(str).str.lower().str.contains(k, regex=False)
        )
        filtered = filtered[mask]
    return filtered


def mode_teacher(courses: pd.DataFrame, refs: pd.DataFrame, refs_with_holdings: pd.DataFrame) -> None:
    st.subheader("教員用モード：同じ図書を使っている授業を探す")
    st.caption("本のタイトル・著者・ISBNで検索し、その本を使っている他の授業を表示します。")
    books = aggregate_books(refs)
    if books.empty:
        st.warning("参考文献が抽出されていません。管理モードでKdBデータを確認してください。")
        return

    query = st.text_input("本を検索", placeholder="例: Quantum Computation / Nielsen / 9781107002173")
    shown = books.copy()
    if query:
        q = query.lower()
        shown = shown[
            shown["title"].astype(str).str.lower().str.contains(q, regex=False)
            | shown["author"].astype(str).str.lower().str.contains(q, regex=False)
            | shown["isbn"].astype(str).str.lower().str.contains(q, regex=False)
        ]

    st.dataframe(
        shown[["title", "author", "isbn", "course_count", "courses", "instructors"]],
        use_container_width=True,
        hide_index=True,
    )

    labels = [f"{r.title} / {r.author} / {r.isbn}" for r in shown.itertuples()]
    if labels:
        selected_label = st.selectbox("詳細を見る図書", labels)
        selected_key = shown.iloc[labels.index(selected_label)]["canonical_key"]
        detail = refs_with_holdings[refs_with_holdings["canonical_key"] == selected_key].copy()
        st.markdown("#### この図書を使っている授業")
        cols = [
            "year",
            "course_number",
            "course_name",
            "instructor",
            "organization",
            "term",
            "weekday_period",
            "holding_status",
            "cinii_url",
        ]
        st.dataframe(detail[[c for c in cols if c in detail.columns]], use_container_width=True, hide_index=True)


def mode_student(courses: pd.DataFrame, refs_with_holdings: pd.DataFrame) -> None:
    st.subheader("学生用モード：履修予定科目の参考文献が図書館にあるか確認する")
    st.caption("KdBっぽいなにかを参考に、科目を絞り込んで選択し、必要な本の所蔵状況を確認します。")
    filtered = filter_courses_ui(courses, prefix="student_")
    st.write(f"候補科目: {len(filtered)} 件")

    options = [
        f"{r.course_id}｜{r.course_name}｜{r.instructor}｜{r.term}"
        for r in filtered.itertuples()
    ]
    selected = st.multiselect("履修予定の科目を選択", options)
    selected_ids = [s.split("｜", 1)[0] for s in selected]

    selected_courses = courses[courses["course_id"].isin(selected_ids)]
    st.markdown("#### 選択した科目")
    st.dataframe(
        selected_courses[["year", "course_number", "course_name", "instructor", "term", "weekday_period", "credits"]],
        use_container_width=True,
        hide_index=True,
    )

    if selected_ids:
        selected_refs = refs_with_holdings[refs_with_holdings["course_id"].isin(selected_ids)].copy()
        st.markdown("#### 必要な教材・参考文献と所蔵状況")
        display_cols = [
            "course_name",
            "title",
            "author",
            "isbn",
            "material_type",
            "holding_status",
            "holding_confidence",
            "cinii_url",
            "source_text",
        ]
        st.dataframe(selected_refs[[c for c in display_cols if c in selected_refs.columns]], use_container_width=True, hide_index=True)
        st.download_button(
            "選択科目の文献リストをCSVで保存",
            data=csv_bytes(selected_refs),
            file_name="selected_course_references.csv",
            mime="text/csv",
        )
    else:
        st.info("科目を選択すると、参考文献と所蔵状況が表示されます。")


def mode_admin(courses: pd.DataFrame, refs: pd.DataFrame, refs_with_holdings: pd.DataFrame, appid: str, kid: str, fano: str) -> pd.DataFrame:
    st.subheader("管理モード：全リストの一覧・出力・所蔵照合")
    m1, m2, m3, m4 = st.columns(4)
    m1.metric("科目数", f"{len(courses):,}")
    m2.metric("抽出文献数", f"{len(refs):,}")
    m3.metric("図書候補", f"{len(refs[~refs['material_type'].isin(['配付資料','Web資料'])]):,}" if not refs.empty else "0")
    m4.metric("重複除去後の本", f"{refs['canonical_key'].nunique():,}" if not refs.empty else "0")

    with st.expander("科目データ", expanded=False):
        st.dataframe(courses, use_container_width=True, hide_index=True)

    st.markdown("#### 全参考文献リスト")
    filtered_courses = filter_courses_ui(courses, prefix="admin_")
    filtered_refs = refs_with_holdings[refs_with_holdings["course_id"].isin(filtered_courses["course_id"])].copy()
    kind_filter = st.multiselect(
        "資料種別",
        sorted(filtered_refs["material_type"].dropna().unique()) if not filtered_refs.empty else [],
        default=[],
    )
    if kind_filter:
        filtered_refs = filtered_refs[filtered_refs["material_type"].isin(kind_filter)]
    st.dataframe(filtered_refs, use_container_width=True, hide_index=True)

    c1, c2 = st.columns(2)
    with c1:
        st.download_button(
            "全リストCSVを保存",
            data=csv_bytes(filtered_refs),
            file_name="tsukuba_bookmap_all_references.csv",
            mime="text/csv",
        )
    with c2:
        st.download_button(
            "全リストExcelを保存",
            data=excel_bytes({"courses": courses, "references": filtered_refs}),
            file_name="tsukuba_bookmap_all_references.xlsx",
            mime="application/vnd.openxmlformats-officedocument.spreadsheetml.sheet",
        )

    st.markdown("#### CiNii Books APIで所蔵照合")
    st.caption("大量照合時は大学・CiNii側への負荷を避けるため、件数を絞り、キャッシュを使ってください。")
    max_lookup = st.number_input("今回照合する上限件数", min_value=1, max_value=500, value=30, step=10)
    if st.button("未照合の本を照合する"):
        book_refs = refs[~refs["material_type"].isin(["配付資料", "Web資料"])].drop_duplicates("canonical_key")
        with st.spinner("CiNii Books APIを照合中..."):
            holdings = lookup_references(book_refs, appid=appid, kid=kid, fano=fano, limit=int(max_lookup))
        st.session_state["holdings"] = holdings
        st.success("照合が完了しました。")
        st.dataframe(holdings, use_container_width=True, hide_index=True)

    return st.session_state.get("holdings", pd.DataFrame())


def main() -> None:
    st.title("📚 Tsukuba BookMap")
    st.caption("KdBシラバス参考文献 × 筑波大学図書館所蔵チェック")

    courses = load_courses_from_sidebar()
    appid, kid, fano = sidebar_settings()
    refs = cached_extract(courses)

    if "holdings" not in st.session_state:
        st.session_state["holdings"] = pd.DataFrame()

    refs_with_holdings = merge_holdings(refs, st.session_state["holdings"])

    tab_teacher, tab_student, tab_admin = st.tabs(["教員用", "学生用", "管理"])
    with tab_teacher:
        mode_teacher(courses, refs, refs_with_holdings)
    with tab_student:
        mode_student(courses, refs_with_holdings)
    with tab_admin:
        new_holdings = mode_admin(courses, refs, refs_with_holdings, appid, kid, fano)
        if not new_holdings.empty:
            refs_with_holdings = merge_holdings(refs, new_holdings)


if __name__ == "__main__":
    main()
